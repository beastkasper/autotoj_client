## 4. Навигация и интеграция с App

### AppView типы для бортжурнала

```typescript
type AppView = '...' | 'logbook' | 'logbook-create' | 'logbook-detail' | 'blog' | '...';
```

### State в App.tsx

```typescript
const [activeTab, setActiveTab] = useState<AppView>('search');
const [selectedLogbookPostId, setSelectedLogbookPostId] = useState<string | null>(null);
const [previousTab, setPreviousTab] = useState<string>('');
```

### Рендер в App.tsx

```tsx
{/* Десктоп — вкладка «Блог» в хедере показывает DesktopBlogPage */}
{activeTab === 'blog' && (
  <DesktopBlogPage
    onCreate={() => {
      setPreviousTab('blog');
      setActiveTab('logbook-create');
    }}
    onPostClick={(postId) => {
      setPreviousTab('blog');
      setSelectedLogbookPostId(postId);
      setActiveTab('logbook-detail');
    }}
  />
)}

{/* Мобильный — список бортжурнала (из меню) */}
{activeTab === 'logbook' && (
  <LogbookListPage
    onBack={() => setActiveTab('menu')}
    onCreate={() => setActiveTab('logbook-create')}
    onPostClick={(postId) => {
      setSelectedLogbookPostId(postId);
      setActiveTab('logbook-detail');
    }}
  />
)}

{/* Создание поста (общий для мобильного и десктопа) */}
{activeTab === 'logbook-create' && (
  <LogbookCreatePage
    onClose={() => {
      setActiveTab(previousTab === 'blog' ? 'blog' : 'logbook');
    }}
    onSubmit={(data) => {
      console.log('Logbook post created:', data);
      setActiveTab(previousTab === 'blog' ? 'blog' : 'logbook');
    }}
  />
)}

{/* Детали поста — разный компонент в зависимости от previousTab */}
{activeTab === 'logbook-detail' && selectedLogbookPostId && (
  previousTab === 'blog' ? (
    <DesktopBlogDetailPage
      postId={selectedLogbookPostId}
      onBack={() => setActiveTab('blog')}
    />
  ) : (
    <LogbookDetailPage
      postId={selectedLogbookPostId}
      onBack={() => setActiveTab('logbook')}
    />
  )
)}
```

### Навигация из меню

```tsx
// В MenuPage.tsx — кнопка "Бортжурнал"
<button onClick={() => onNavigate('logbook')}>
  Бортжурнал
</button>
```

### Навигация из MyActivityPage

```tsx
// Вкладка "Бортжурнал" показывает посты текущего пользователя
onLogbookPostClick={(postId) => {
  setSelectedLogbookPostId(postId);
  setActiveTab('logbook-detail');
}}
onLogbookCreate={() => setActiveTab('logbook-create')}
```

const handleShare = async () => {
  const shareData = {
    title: post.title,
    text: post.text.substring(0, 100) + '...',
    url: window.location.href,
  };

  try {
    if (navigator.share && navigator.canShare?.(shareData)) {
      await navigator.share(shareData);
    } else {
      // fallback: копирование в буфер через textarea
      const textArea = document.createElement('textarea');
      textArea.value = window.location.href;
      textArea.style.position = 'fixed';
      textArea.style.left = '-999999px';
      document.body.appendChild(textArea);
      textArea.focus();
      textArea.select();
      document.execCommand('copy');
      document.body.removeChild(textArea);
      toast.success('Ссылка скопирована');
    }
  } catch (error) {
    // fallback copy
  }
};

              <input type="text" value={title} onChange={(e) => setTitle(e.target.value)}
                placeholder="О чём вы хотите рассказать?"
                className="w-full px-4 py-3 bg-background border border-border rounded-lg
                  focus:outline-none focus:ring-2 focus:ring-primary" />
            </div>
            <div>
              <label className="block text-sm font-medium mb-2">Тема</label>
              <button onClick={() => setShowCategorySheet(true)}
                className="w-full flex items-center justify-between px-4 py-3
                  bg-background border border-border rounded-lg hover:bg-muted/50">
                <span className={selectedCategory === 'Без темы' ? 'text-muted-foreground' : ''}>
                  {selectedCategory}
                </span>
                <ChevronRight className="w-5 h-5 text-muted-foreground" />
              </button>
            </div>
            <div>
              <label className="block text-sm font-medium mb-2">Текст</label>
              <textarea value={text} onChange={(e) => setText(e.target.value)}
                placeholder="Расскажите подробнее..." rows={6}
                className="w-full px-4 py-3 bg-background border border-border rounded-lg
                  focus:outline-none focus:ring-2 focus:ring-primary resize-none" />
            </div>
            <div className="flex gap-3">
              <button className="flex-1 flex items-center justify-center gap-2 px-4 py-3
                bg-background border border-border rounded-lg hover:bg-muted/50">
                <ImageIcon className="w-5 h-5 text-muted-foreground" />
                <span className="text-sm font-medium">Фото</span>
                <span className="text-xs text-muted-foreground">(до 10)</span>

                bg-background border border-border rounded-lg hover:bg-muted/50">
                <Video className="w-5 h-5 text-muted-foreground" />

            <p className="text-xs text-muted-foreground">
              Пожалуйста, соблюдайте правила сообщества и будьте вежливы к другим пользователям.
            </p>
          </div>
        </div>
      </div>

      {/* Category Sheet */}
      {showCategorySheet && (
        <div className="fixed inset-0 bg-black/50 z-[60] flex items-end"
          onClick={() => setShowCategorySheet(false)}>
          <div className="w-full max-w-md mx-auto bg-card rounded-t-3xl max-h-[70vh]
            overflow-hidden flex flex-col" onClick={(e) => e.stopPropagation()}>
            <div className="flex justify-center py-2">
              <div className="w-10 h-1 bg-muted rounded-full" />
            </div>
            <div className="flex items-center justify-between p-4 border-b border-border">
              <h2 className="font-semibold">Выберите тему</h2>
              <button onClick={() => setShowCategorySheet(false)}
                className="p-2 -mr-2 hover:bg-muted rounded-full">
                <X className="w-5 h-5" />
              </button>
            </div>
            <div className="flex-1 overflow-y-auto">
              {categories.map((category) => (
                <button key={category}
                  onClick={() => { setSelectedCategory(category); setShowCategorySheet(false); }}
                  className={`w-full text-left px-4 py-4 border-b border-border
                    hover:bg-muted/50 transition-colors ${
                    selectedCategory === category ? 'bg-primary/5 text-primary font-medium' : ''
                  }`}>
                  {category}
                </button>
              ))}
            </div>
          </div>
        </div>
      )}
    </>
  );
}
