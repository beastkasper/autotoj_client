## 3. Навигация и интеграция с App

### Флоу навигации

```
App.tsx
  │
  ├─ activeTab === 'parts' → <PartsPage>          (список запчастей)
  ├─ activeTab === 'parts-detail' → <PartsDetailPage>  (детали запчасти)
  └─ activeTab === 'post' → <PostAdPage>           (добавление объявления)
       │
       ├─ currentStep === 'category' → <CategorySelectSheet>
       │     └─ user selects 'parts'
       │
       ├─ selectedCategory === 'parts' → <PartsListingPlaceholder>
       │     └─ user selects subcategory (e.g. 'Шины')
       │
       ├─ showTiresForm → <TiresListingForm>
       │     └─ validated → <TiresPreviewStep>
       │           └─ published → <PublishSuccessScreen>
       │
       └─ (аналогично для остальных 8 форм)
```

### 4.1 BottomSheetSelect

```tsx
// src/components/form/BottomSheetSelect.tsx
import { X, Check, Plus } from 'lucide-react';
import { useState } from 'react';

interface BottomSheetSelectProps {
  value: string;
  options: string[];
  onChange: (value: string) => void;
  placeholder?: string;
  allowCustom?: boolean;
  onAddCustom?: () => void;
  customButtonText?: string;
  error?: string;
  title?: string;
}

export function BottomSheetSelect({
  value, options, onChange, placeholder, allowCustom,
  onAddCustom, customButtonText, error, title,
}: BottomSheetSelectProps) {
  const [isOpen, setIsOpen] = useState(false);

  const handleSelect = (option: string) => {
    onChange(option);
    setIsOpen(false);
  };

  return (
    <>
      {/* Trigger Button */}
      <button
        onClick={() => setIsOpen(true)}
        className={`w-full h-12 px-4 rounded-[12px] text-[15px] text-left transition-colors
          flex items-center justify-between ${
          error
            ? 'bg-white border border-[#E53935]'
            : 'bg-white border border-[#C7C7CC]'
        } ${!value && 'text-[#8E8E93]'}`}
        style={{ fontFamily: 'Manrope, system-ui, sans-serif' }}
      >
        <span>{value || placeholder}</span>
        <svg className="w-5 h-5" fill="none" viewBox="0 0 24 24"
          stroke="currentColor" strokeWidth={1.5}>
          <path strokeLinecap="round" strokeLinejoin="round" d="M19 9l-7 7-7-7" />
        </svg>
      </button>

      {/* Bottom Sheet Modal */}
      {isOpen && (
        <>
          <div
            className="fixed inset-0 bg-black/10 z-[60]"
            onClick={() => setIsOpen(false)}
          />
          <div className="fixed bottom-0 left-0 right-0 bg-white rounded-t-[24px]
            shadow-xl z-[70] max-h-[80vh] flex flex-col animate-slide-up">
            {/* Header */}
            <div className="flex-shrink-0 flex items-center justify-between
              px-4 py-4 border-b border-[#E5E5EA]">
              <div className="w-10" />
              <h2 className="text-[17px] font-semibold"
                style={{ fontFamily: 'Manrope, system-ui, sans-serif' }}>
                {title || placeholder || 'Выберите значение'}
              </h2>
              <button onClick={() => setIsOpen(false)}
                className="w-10 h-10 -mr-2 flex items-center justify-center">
                <X className="w-6 h-6" strokeWidth={1.5} />
              </button>
            </div>

            {/* Options List */}
            <div className="flex-1 overflow-y-auto">
              {options.map((option, index) => (
                <div key={option}>
                  <button
                    onClick={() => handleSelect(option)}
                    className="w-full px-4 py-3 flex items-center justify-between
                      hover:bg-[#F2F2F7] transition-colors"
                  >
                    <span className="text-[15px] text-left"
                      style={{ fontFamily: 'Manrope, system-ui, sans-serif' }}>
                      {option}
                    </span>
                    {value === option && (
                      <Check className="w-5 h-5 text-[#000000]" strokeWidth={2} />
                    )}
                  </button>
                  {index < options.length - 1 && (
                    <div className="h-px bg-[#E5E5EA] mx-4" />
                  )}
                </div>
              ))}

              {/* Add Custom Button */}
              {allowCustom && onAddCustom && (
                <>
                  {options.length > 0 && <div className="h-px bg-[#E5E5EA] mx-4" />}
                  <button
                    onClick={() => { setIsOpen(false); onAddCustom(); }}
                    className="w-full px-4 py-3 flex items-center gap-2
                      hover:bg-[#F2F2F7] transition-colors"
                  >
                    <Plus className="w-5 h-5" strokeWidth={1.5} />
                    <span className="text-[15px]"
                      style={{ fontFamily: 'Manrope, system-ui, sans-serif' }}>
                      {customButtonText || 'Добавить'}
                    </span>
                  </button>
                </>
              )}
            </div>
            <div className="h-8 flex-shrink-0" />
          </div>
        </>
      )}
    </>
  );
}
```

### 4.3 AddValueModal

```tsx
// src/components/form/AddValueModal.tsx
import { useState } from 'react';
import { X } from 'lucide-react';

interface AddValueModalProps {
  isOpen: boolean;
  onClose: () => void;
  onAdd: (value: string) => void;
  title: string;
  placeholder?: string;  // default: 'Введите значение'
  buttonText?: string;   // default: 'Добавить'
}

export function AddValueModal({
  isOpen, onClose, onAdd, title,
  placeholder = 'Введите значение', buttonText = 'Добавить',
}: AddValueModalProps) {
  const [inputValue, setInputValue] = useState('');

  const handleAdd = () => {
    if (inputValue.trim()) {
      onAdd(inputValue.trim());
      setInputValue('');
      onClose();
    }
  };

  const handleClose = () => { setInputValue(''); onClose(); };

  if (!isOpen) return null;

  return (
    <>
      <div className="fixed inset-0 bg-black/40 z-60 transition-opacity duration-200"
        onClick={handleClose} />
      <div className="fixed inset-0 flex items-end justify-center z-60 pointer-events-none">
        <div className="bg-white rounded-t-[24px] p-6 w-full max-w-md
          animate-slide-up pointer-events-auto shadow-2xl">
          <div className="flex items-center justify-between mb-4">
            <h2 className="text-[18px] font-semibold"
              style={{ fontFamily: 'Manrope, system-ui, sans-serif' }}>{title}</h2>
            <button onClick={handleClose} className="w-6 h-6 flex items-center justify-center -mr-2">
              <X className="w-6 h-6" strokeWidth={1.5} />
            </button>
          </div>
          <input type="text" placeholder={placeholder} value={inputValue}
            onChange={(e) => setInputValue(e.target.value)}
            onKeyPress={(e) => e.key === 'Enter' && handleAdd()}
            className="w-full h-12 px-4 bg-white rounded-[12px] border border-[#E5E5EA]
              text-[15px] placeholder:text-[#9E9E9E] focus:outline-none
              focus:border-2 focus:border-[#000000]"
            style={{ fontFamily: 'Manrope, system-ui, sans-serif' }}
            autoFocus
          />
          <button onClick={handleAdd} disabled={!inputValue.trim()}
            className={`w-full h-12 mt-4 rounded-[12px] text-[16px] font-semibold
              transition-colors ${inputValue.trim()
              ? 'bg-[#111111] text-white active:bg-[#000000]'
              : 'bg-[#E5E5EA] text-[#9E9E9E] cursor-not-allowed'}`}
            style={{ fontFamily: 'Manrope, system-ui, sans-serif' }}>
            {buttonText}
          </button>
        </div>
      </div>
    </>
  );
}
```

```tsx
// src/components/form/EditContactsModal.tsx
import { useState, useEffect } from 'react';
import { X } from 'lucide-react';

interface EditContactsModalProps {
  isOpen: boolean;
  onClose: () => void;
  currentName: string;
  currentPhone: string;
  currentCity: string;
  onSave: (phone: string, city: string) => void;
}

export function EditContactsModal({
  isOpen, onClose, currentName, currentPhone, currentCity, onSave
}: EditContactsModalProps) {
  const [phone, setPhone] = useState(currentPhone || '');
  const [city, setCity] = useState(currentCity || '');

  useEffect(() => {
    if (isOpen) { setPhone(currentPhone || ''); setCity(currentCity || ''); }
  }, [isOpen, currentPhone, currentCity]);

  if (!isOpen) return null;

  const formatPhone = (value: string) => {
    const digits = value.replace(/\D/g, '').slice(0, 9);
    if (digits.length <= 2) return digits;
    if (digits.length <= 5) return `${digits.slice(0, 2)} ${digits.slice(2)}`;
    if (digits.length <= 7) return `${digits.slice(0, 2)} ${digits.slice(2, 5)} ${digits.slice(5)}`;
    return `${digits.slice(0, 2)} ${digits.slice(2, 5)} ${digits.slice(5, 7)} ${digits.slice(7)}`;
  };

  return (
    <div className="fixed inset-0 z-[100] flex items-end justify-center" onClick={onClose}>
      <div className="absolute inset-0 bg-black/40 backdrop-blur-sm" />
      <div className="relative w-full max-w-[500px] bg-white rounded-t-[24px] shadow-2xl"
        onClick={(e) => e.stopPropagation()} style={{ maxHeight: '90vh' }}>
        {/* Header */}
        <div className="flex items-center justify-between px-5 py-4 border-b border-[#E5E5EA]">
          <h2 className="text-[17px] font-semibold text-[#000000]"
            style={{ fontFamily: 'Manrope, system-ui, sans-serif' }}>Изменить контакты</h2>
          <button onClick={onClose}
            className="w-8 h-8 flex items-center justify-center rounded-full
              hover:bg-[#F2F2F7] active:opacity-60 transition-all">
            <X className="w-5 h-5 text-[#000000]" />
          </button>
        </div>
        <div className="px-5 py-6 space-y-5">
          {/* Имя (read-only) */}
          <div>
            <label className="block text-[14px] font-medium text-[#000000] mb-2"
              style={{ fontFamily: 'Manrope, system-ui, sans-serif' }}>Имя</label>
            <input type="text" value={currentName} disabled
              className="w-full h-12 px-4 bg-[#F5F5F5] border border-[#E5E5EA]
                rounded-[12px] text-[15px] text-[#8E8E93] cursor-not-allowed"
              style={{ fontFamily: 'Manrope, system-ui, sans-serif' }} />
            <p className="mt-1.5 text-[12px] text-[#8E8E93]"
              style={{ fontFamily: 'Manrope, system-ui, sans-serif' }}>
              Имя можно изменить в разделе Профиль
            </p>
          </div>
          {/* Телефон */}
          <div>
            <label className="block text-[14px] font-medium text-[#000000] mb-2"
              style={{ fontFamily: 'Manrope, system-ui, sans-serif' }}>
              Номер телефона <span className="text-[#FF3B30]">*</span>
            </label>
            <div className="flex gap-2">
              <input type="text" value="+992" disabled
                className="w-20 h-12 px-4 rounded-[12px] text-[15px] bg-[#F5F5F5]
                  text-[#000000] font-medium border border-[#E5E5EA]"
                style={{ fontFamily: 'Manrope, system-ui, sans-serif' }} />
              <input type="tel" placeholder="900 00 00" value={phone}
                onChange={(e) => setPhone(formatPhone(e.target.value))}
                className="flex-1 h-12 px-4 bg-white border border-[#C7C7CC] rounded-[12px]
                  text-[15px] focus:outline-none focus:border-2 focus:border-[#000000]"
                style={{ fontFamily: 'Manrope, system-ui, sans-serif' }} />
            </div>
          </div>
          {/* Город */}
          <div>
            <label className="block text-[14px] font-medium text-[#000000] mb-2"
              style={{ fontFamily: 'Manrope, system-ui, sans-serif' }}>
              Город <span className="text-[#FF3B30]">*</span>
            </label>
            <input type="text" placeholder="Душанбе" value={city}
              onChange={(e) => setCity(e.target.value)}
              className="w-full h-12 px-4 bg-white border border-[#C7C7CC] rounded-[12px]
                text-[15px] focus:outline-none focus:border-2 focus:border-[#000000]"
              style={{ fontFamily: 'Manrope, system-ui, sans-serif' }} />
          </div>
        </div>
        <div className="px-5 pb-6 space-y-3">
          <button onClick={() => { onSave(phone, city); onClose(); }}
            disabled={!phone || !city}
            className="w-full h-[52px] rounded-[14px] font-semibold bg-[#000000] text-white
              disabled:opacity-40 disabled:cursor-not-allowed active:opacity-70 transition-opacity"
            style={{ fontFamily: 'Manrope, system-ui, sans-serif' }}>Сохранить</button>
          <button onClick={onClose}
            className="w-full h-[52px] rounded-[14px] font-semibold bg-[#F2F2F7]
              text-[#000000] active:opacity-70 transition-opacity"
            style={{ fontFamily: 'Manrope, system-ui, sans-serif' }}>Отмена</button>
        </div>
```

```typescript
export const userStore = {

    if (typeof window === 'undefined') return defaultUser;

    if (stored) {

    this.saveProfile(defaultUser);

  },
```

```typescript
export function formatDateWithCity(date: Date | string | undefined, city: string): string {
  if (!date) date = new Date();
  const d = typeof date === 'string' ? new Date(date) : date;
  if (isNaN(d.getTime())) return `${city}`;
  return `${d.getDate()} ${months[d.getMonth()]} ${d.getFullYear()}, ${city}`;
}

export function formatDate(date: Date | string): string {
  const d = typeof date === 'string' ? new Date(date) : date;
  return `${d.getDate()} ${months[d.getMonth()]} ${d.getFullYear()}`;
}
```

## 17. Превью-шаги

### Props (одинаковые)

```typescript

}
```

### Общая структура превью

```
┌──────────────────────────────────────┐
│ HEADER: ← [Предпросмотр] [X]        │
├──────────────────────────────────────┤
│ ФОТО (если есть)                     │
│ grid grid-cols-3 gap-2               │
│ или hero 1 фото aspect-[4/3]        │
├──────────────────────────────────────┤
│ ЦЕНА: text-[20px] font-bold         │
│ + "сомони"                           │
├──────────────────────────────────────┤
│ InfoRow (label → value) пары:        │
│ border-b border-[#F2F2F7]           │
│ label: text-[#8E8E93]               │
│ value: text-[#000000]               │
├──────────────────────────────────────┤
│ ОПИСАНИЕ (если есть)                 │
├──────────────────────────────────────┤
│ КОНТАКТЫ: Имя, +992 xxx, Город      │
├──────────────────────────────────────┤
│ КНОПКА: "Опубликовать"              │
│ bg-[#111111] text-white             │
└──────────────────────────────────────┘
```

### InfoRow компонент (используется внутри превью)

```tsx
function InfoRow({ label, value }: { label: string; value: string }) {
  return (
    <div className="flex justify-between py-3 border-b border-[#F2F2F7]">
      <span className="text-[15px] text-[#8E8E93]"
        style={{ fontFamily: 'Manrope, system-ui, sans-serif' }}>{label}</span>
      <span className="text-[15px] text-[#000000] font-medium text-right"
        style={{ fontFamily: 'Manrope, system-ui, sans-serif' }}>{value}</span>
    </div>
  );
}
```
